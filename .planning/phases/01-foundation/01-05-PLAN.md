    /**

     * 2. Never call the Anthropic client outside this file.
     * 3. All AI output shapes are validated by Zod schemas defined here.
     *
     * Phase 1: Stubs only. Schemas are defined so Phase 5 can implement
     * without changing call sites or output types.
     */
    import "server-only";
    import Anthropic from "@anthropic-ai/sdk";
    import { z } from "zod";

    // ---------------------------------------------------------------------------
    // Client (instantiated at module level; not called in Phase 1)
    // ---------------------------------------------------------------------------
    const client = new Anthropic({
      apiKey: process.env.ANTHROPIC_API_KEY ?? "",
    });

    // ---------------------------------------------------------------------------
    // Output Schemas — define structured output contracts for v2 features
    // ---------------------------------------------------------------------------

    export const AnomalyTypeSchema = z.enum([
      "capacity_drop",
      "ce_anomaly",
      "voltage_plateau",
      "resistance_increase",
    ]);

    export const SeveritySchema = z.enum(["low", "medium", "high"]);

    export const AnomalySchema = z.object({
      anomalies: z.array(
        z.object({
          cycleNumber: z.number().int().positive(),
          type: AnomalyTypeSchema,
          severity: SeveritySchema,
          description: z.string().min(1),
        })
      ),
      summary: z.string().min(1),
    });

    export type AnomalyResult = z.infer<typeof AnomalySchema>;
    export type AnomalyType = z.infer<typeof AnomalyTypeSchema>;
    export type Severity = z.infer<typeof SeveritySchema>;

    export const TestSummarySchema = z.object({
      summary: z.string().min(1),
      keyFindings: z.array(z.string()),
      recommendedNextSteps: z.array(z.string()),
    });

    export type TestSummaryResult = z.infer<typeof TestSummarySchema>;

    // ---------------------------------------------------------------------------
    // Exported Functions — stubs until Phase 5
