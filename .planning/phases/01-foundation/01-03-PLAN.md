  <name>Task 2: Create auth helper with user-sync-to-DB utility</name>
  <files>src/lib/auth.ts</files>
  <action>
    Create `src/lib/auth.ts` (server-only):
    ```typescript
    import "server-only";
    import { auth, currentUser } from "@clerk/nextjs/server";
    import { prisma } from "@/lib/prisma";

    /**
     * requireAuth — call at the top of any Server Component or Server Action that
     * requires authentication. Returns the Clerk userId.
     * Throws if not authenticated (Clerk middleware should prevent this, but belt-and-suspenders).
     */
    export async function requireAuth(): Promise<string> {
      const { userId } = await auth();
      if (!userId) throw new Error("Unauthorized");
      return userId;
    }

    /**
     * syncUserToDb — upsert the signed-in Clerk user into the local users table.
     * Call this on first sign-in (e.g., from a Server Action triggered after sign-in).
     * Safe to call repeatedly — upsert is idempotent.
     */
    export async function syncUserToDb(): Promise<void> {
      const clerkUser = await currentUser();
      if (!clerkUser) throw new Error("No authenticated user");

      const email = clerkUser.emailAddresses[0]?.emailAddress ?? "";
      const name = `${clerkUser.firstName ?? ""} ${clerkUser.lastName ?? ""}`.trim() || email;

      await prisma.user.upsert({
        where: { id: clerkUser.id },
        update: { email, name },
        create: { id: clerkUser.id, email, name },
      });
    }
    ```

    Wire `syncUserToDb` to run after sign-in by calling it from `src/app/(auth)/dashboard/page.tsx`:
    ```typescript
    import { syncUserToDb } from "@/lib/auth";
    export default async function DashboardPage() {
      await syncUserToDb();
      return <div className="p-8 text-zinc-100"><h1>Dashboard</h1></div>;
    }
    ```
    This runs server-side on every dashboard load — the upsert is idempotent so subsequent calls after first sync are no-ops.

    The `requireAuth` function establishes the pattern all future phases use for server-side auth checks on write paths.
